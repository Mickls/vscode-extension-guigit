# Changes Commit Workflow Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the right-panel `Changes` workflow with staged changes, unstaged changes, stash management, commit execution, and AI-generated commit messages.

**Architecture:** The extension backend owns Git status parsing, staging, unstaging, discard confirmation, stash operations, commit execution, diff opening, settings, secrets, provider calls, and refresh notifications. The Webview renders typed ViewModels and sends intent-only RPC messages from a new `Details / Changes` tab surface.

**Tech Stack:** pnpm, TypeScript, React, Vitest, VS Code Extension API, simple-git, VS Code Language Model API, OpenAI-compatible chat completions.

```ts
import type { StashEntryViewModel, WorkingTreeFileChangeViewModel } from "../rpc/contract";

export interface ParsedWorkingTreeStatus {
  staged: readonly WorkingTreeFileChangeViewModel[];
  unstaged: readonly WorkingTreeFileChangeViewModel[];
}

export function parsePorcelainStatus(output: string): ParsedWorkingTreeStatus {
  const staged: WorkingTreeFileChangeViewModel[] = [];
  const unstaged: WorkingTreeFileChangeViewModel[] = [];

  for (const line of output.split("\n").filter((item) => item.length > 0)) {
    const indexStatus = line[0]!;
    const workTreeStatus = line[1]!;
    const path = line.slice(3);

    if (line.startsWith("?? ")) {
      unstaged.push(createWorkingTreeFile(path, "untracked", "added"));
      continue;
    }

    if (indexStatus !== " ") {
      staged.push(createWorkingTreeFile(path, "staged", statusFromCode(indexStatus)));
    }

    if (workTreeStatus !== " ") {
      unstaged.push(createWorkingTreeFile(path, "unstaged", statusFromCode(workTreeStatus)));
    }
  }

  return { staged, unstaged };
}

export function parseStashList(output: string): readonly StashEntryViewModel[] {
  return output
    .split("\n")
    .filter((line) => line.trim().length > 0)
    .map((line) => {
      const [ref, ...messageParts] = line.split(": ");
      const message = messageParts.join(": ");
      return {
        branch: branchFromStashMessage(message),
        date: "",
        message,
        ref: ref!
      };
    });
}

function createWorkingTreeFile(
  path: string,
  area: WorkingTreeFileChangeViewModel["area"],
  status: WorkingTreeFileChangeViewModel["status"]
): WorkingTreeFileChangeViewModel {
  return {
    area,
    binary: false,
    deletions: 0,
    insertions: 0,
    path,
    status
  };
}

function statusFromCode(code: string): WorkingTreeFileChangeViewModel["status"] {
  if (code === "A") {
    return "added";
  }

  if (code === "D") {
    return "deleted";
  }

  if (code === "R") {
    return "renamed";
  }

  if (code === "C") {
    return "copied";
  }

  return "modified";
}

function branchFromStashMessage(message: string): string {
  const match = /(?:WIP on|On) ([^:]+):/.exec(message);
  return match?.[1] ?? "";
}
```

```ts
import { simpleGit } from "simple-git";
import type { OperationResultViewModel, WorkingTreeViewModel } from "../rpc/contract";
import { parsePorcelainStatus, parseStashList } from "./WorkingTreeParser";

export interface WorkingTreeServiceInput {
  gitRaw?: (repositoryRoot: string, args: readonly string[]) => Promise<string>;
}

export class WorkingTreeService {
  private readonly gitRaw: (repositoryRoot: string, args: readonly string[]) => Promise<string>;

  public constructor(input: WorkingTreeServiceInput = {}) {
    this.gitRaw = input.gitRaw ?? ((repositoryRoot, args) => simpleGit(repositoryRoot).raw([...args]));
  }

  public async load(repositoryId: string, repositoryRoot: string): Promise<WorkingTreeViewModel> {
    const [branchOutput, statusOutput, stashOutput] = await Promise.all([
      this.gitRaw(repositoryRoot, ["rev-parse", "--abbrev-ref", "HEAD"]),
      this.gitRaw(repositoryRoot, ["status", "--porcelain=v1"]),
      this.gitRaw(repositoryRoot, ["stash", "list"])
    ]);
    const parsedStatus = parsePorcelainStatus(statusOutput);

    return {
      branch: branchOutput.trim(),
      repositoryId,
      repositoryRoot,
      staged: parsedStatus.staged,
      stashes: parseStashList(stashOutput),
      unstaged: parsedStatus.unstaged
    };
  }
}
```
